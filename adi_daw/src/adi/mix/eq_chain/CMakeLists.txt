if(TARGET adi_mix_eq_chain)
    return()
endif()

add_library(adi_mix_eq_chain STATIC Core.cpp)
set_target_properties(adi_mix_eq_chain PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_eq_chain PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_eq_chain PUBLIC cxx_std_20)
target_link_libraries(adi_mix_eq_chain PRIVATE adi_mix_eq adi_mix_carve adi_mix_focus)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_eq_chain PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_eq_chain PRIVATE /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_eq_chain PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_eq_chain PRIVATE -ffp-contract=off)
endif()

find_package(Threads REQUIRED)
add_library(adi_mix_eq_chain_backend STATIC Backend.cpp)
set_target_properties(adi_mix_eq_chain_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_eq_chain_backend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_eq_chain_backend PUBLIC cxx_std_20)
target_link_libraries(adi_mix_eq_chain_backend PUBLIC adi_mix_eq_chain PRIVATE Threads::Threads)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_eq_chain_backend PRIVATE adi_warnings)
endif()
