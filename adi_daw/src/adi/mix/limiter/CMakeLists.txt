# Shared component for DAW and suite wrappers. No JUCE or plug-in SDK.
if(TARGET adi_mix_limiter)
    return()
endif()
find_package(Threads REQUIRED)
add_library(adi_mix_limiter_dispatch STATIC Dsp/MultirateDispatch.cpp)
target_include_directories(adi_mix_limiter_dispatch PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(adi_mix_limiter_dispatch PRIVATE cxx_std_20)
set_target_properties(adi_mix_limiter_dispatch PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_limiter_dispatch PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_limiter_dispatch PRIVATE /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_limiter_dispatch PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off)
    endif()
    if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(adi_mix_limiter_dispatch PRIVATE /GL-)
    endif()
    include(CheckCXXSourceCompiles)
    check_cxx_source_compiles("#if !defined(_M_X64) || defined(_M_ARM64EC)\n#error Expected x64\n#endif\nint main(){return 0;}" ADI_MIX_LIMITER_X64)
    if(ADI_MIX_LIMITER_X64)
        add_library(adi_mix_limiter_avx OBJECT Dsp/MultirateAvx.cpp)
        target_include_directories(adi_mix_limiter_avx PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_features(adi_mix_limiter_avx PRIVATE cxx_std_20)
        if(TARGET adi_warnings)
            target_link_libraries(adi_mix_limiter_avx PRIVATE adi_warnings)
        endif()
        target_compile_options(adi_mix_limiter_avx PRIVATE /fp:precise /arch:AVX)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
            target_compile_options(adi_mix_limiter_avx PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off)
        endif()
        if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(adi_mix_limiter_avx PRIVATE /GL-)
        endif()
        set_target_properties(adi_mix_limiter_avx PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
        target_sources(adi_mix_limiter_dispatch PRIVATE $<TARGET_OBJECTS:adi_mix_limiter_avx>)
        target_compile_definitions(adi_mix_limiter_dispatch PRIVATE ADIL_HAS_AVX1_KERNELS=1)
    endif()
else()
    target_compile_options(adi_mix_limiter_dispatch PRIVATE -ffp-contract=off)
endif()
add_library(adi_mix_limiter STATIC DspEngine.cpp)
target_include_directories(adi_mix_limiter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_limiter PUBLIC cxx_std_20)
target_link_libraries(adi_mix_limiter PRIVATE adi_mix_limiter_dispatch Threads::Threads)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_limiter PRIVATE adi_warnings)
endif()
set_target_properties(adi_mix_limiter PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(adi_mix_limiter PRIVATE /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_limiter PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_limiter PRIVATE -ffp-contract=off)
endif()
