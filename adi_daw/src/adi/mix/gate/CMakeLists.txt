cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdiMixGate LANGUAGES CXX)
    enable_testing()
endif()
if(TARGET adi_mix_gate)
    return()
endif()
add_library(adi_mix_gate STATIC Core.cpp ControlMapping.cpp)
set_target_properties(adi_mix_gate PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_gate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_gate PUBLIC cxx_std_20)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_gate PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_gate PRIVATE /W4 /WX /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_gate PRIVATE /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_gate PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(adi_mix_gate_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/test_mix_gate.cpp")
    target_link_libraries(adi_mix_gate_tests PRIVATE adi_mix_gate)
    if(MSVC)
        target_compile_options(adi_mix_gate_tests PRIVATE /W4 /WX /fp:precise)
    else()
        target_compile_options(adi_mix_gate_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
    endif()
    add_test(NAME adi_mix_gate_tests COMMAND adi_mix_gate_tests)
endif()
