cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdiMixCompressor LANGUAGES CXX)
    enable_testing()
endif()
if(TARGET adi_mix_compressor)
    return()
endif()
add_library(adi_mix_compressor STATIC Core.cpp)
set_target_properties(adi_mix_compressor PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_compressor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_compressor PUBLIC cxx_std_20)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_compressor PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_compressor PRIVATE /W4 /WX /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_compressor PRIVATE /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_compressor PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(adi_mix_compressor_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/test_mix_compressor.cpp")
    target_link_libraries(adi_mix_compressor_tests PRIVATE adi_mix_compressor)
    if(MSVC)
        target_compile_options(adi_mix_compressor_tests PRIVATE /W4 /WX /fp:precise)
    else()
        target_compile_options(adi_mix_compressor_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
    endif()
    add_test(NAME adi_mix_compressor_tests COMMAND adi_mix_compressor_tests)
endif()
