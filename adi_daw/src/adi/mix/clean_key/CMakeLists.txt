cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdiMixCleanKey LANGUAGES CXX)
    enable_testing()
endif()
if(TARGET adi_mix_clean_key)
    return()
endif()
if(NOT TARGET adi_mix_compressor)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../compressor" "${CMAKE_CURRENT_BINARY_DIR}/compressor")
endif()
if(NOT TARGET adi_mix_key_eq)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../key_eq" "${CMAKE_CURRENT_BINARY_DIR}/key_eq")
endif()
add_library(adi_mix_clean_key STATIC Core.cpp)
set_target_properties(adi_mix_clean_key PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_clean_key PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_clean_key PUBLIC cxx_std_20)
target_link_libraries(adi_mix_clean_key PRIVATE adi_mix_compressor adi_mix_key_eq)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_clean_key PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_clean_key PRIVATE /W4 /WX /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_clean_key PRIVATE /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_clean_key PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(adi_mix_clean_key_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/test_mix_clean_key.cpp")
    target_link_libraries(adi_mix_clean_key_tests PRIVATE adi_mix_clean_key adi_mix_compressor adi_mix_key_eq)
    if(MSVC)
        target_compile_options(adi_mix_clean_key_tests PRIVATE /W4 /WX /fp:precise)
    else()
        target_compile_options(adi_mix_clean_key_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
    endif()
    add_test(NAME adi_mix_clean_key_tests COMMAND adi_mix_clean_key_tests)
endif()
