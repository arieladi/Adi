if(TARGET adi_mix_carve)
    return()
endif()
add_library(adi_mix_carve STATIC Core.cpp)
set_target_properties(adi_mix_carve PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_carve PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_carve PUBLIC cxx_std_20)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_carve PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_carve PRIVATE /fp:strict)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_carve PRIVATE /clang:-ffp-model=strict /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_carve PRIVATE -ffp-contract=off)
endif()
