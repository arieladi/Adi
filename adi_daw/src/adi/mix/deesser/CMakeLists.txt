cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdiMixDeesser LANGUAGES CXX)
    enable_testing()
endif()
if(TARGET adi_mix_deesser)
    return()
endif()
if(NOT TARGET adi_mix_filters)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../filters" "${CMAKE_CURRENT_BINARY_DIR}/filters")
endif()
add_library(adi_mix_deesser STATIC Core.cpp ControlMapping.cpp)
set_target_properties(adi_mix_deesser PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_deesser PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_deesser PUBLIC cxx_std_20)
target_link_libraries(adi_mix_deesser PUBLIC adi_mix_filters)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_deesser PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_deesser PRIVATE /W4 /WX /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_deesser PRIVATE /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_deesser PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(adi_mix_deesser_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/test_mix_deesser.cpp")
    target_link_libraries(adi_mix_deesser_tests PRIVATE adi_mix_deesser)
    if(MSVC)
        target_compile_options(adi_mix_deesser_tests PRIVATE /W4 /WX /fp:precise)
    else()
        target_compile_options(adi_mix_deesser_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -ffp-contract=off)
    endif()
    add_test(NAME adi_mix_deesser_tests COMMAND adi_mix_deesser_tests)
endif()
