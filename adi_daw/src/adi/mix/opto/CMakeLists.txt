cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdiMixOptoGain LANGUAGES CXX)
    enable_testing()
endif()
if(TARGET adi_mix_opto_gain)
    return()
endif()
add_library(adi_mix_opto_gain STATIC GainEnvelope.cpp)
target_include_directories(adi_mix_opto_gain PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_opto_gain PUBLIC cxx_std_20)
set_target_properties(adi_mix_opto_gain PROPERTIES POSITION_INDEPENDENT_CODE ON INTERPROCEDURAL_OPTIMIZATION FALSE)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_opto_gain PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_opto_gain PRIVATE /W4 /WX /fp:precise)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_opto_gain PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off /clang:-fno-lto)
    else()
        target_compile_options(adi_mix_opto_gain PRIVATE /GL-)
    endif()
else()
    target_compile_options(adi_mix_opto_gain PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -fno-fast-math -ffp-contract=off -fno-lto)
endif()
set(ADI_OPTO_GAIN_TEST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests")
function(adi_add_opto_gain_tests)
    add_executable(adi_mix_opto_gain_tests "${ADI_OPTO_GAIN_TEST_ROOT}/test_mix_opto_gain.cpp" "${ADI_OPTO_GAIN_TEST_ROOT}/test_mix_opto_gain_fast.cpp")
    target_link_libraries(adi_mix_opto_gain_tests PRIVATE adi_mix_opto_gain)
    target_compile_features(adi_mix_opto_gain_tests PRIVATE cxx_std_20)
    set_target_properties(adi_mix_opto_gain_tests PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(TARGET adi_warnings)
        target_link_libraries(adi_mix_opto_gain_tests PRIVATE adi_warnings)
    endif()
    if(MSVC)
        target_compile_options(adi_mix_opto_gain_tests PRIVATE /W4 /WX /fp:precise)
        set_property(SOURCE "${ADI_OPTO_GAIN_TEST_ROOT}/test_mix_opto_gain_fast.cpp" APPEND PROPERTY COMPILE_OPTIONS /fp:fast)
    else()
        target_compile_options(adi_mix_opto_gain_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror -fno-fast-math -ffp-contract=off)
        set_property(SOURCE "${ADI_OPTO_GAIN_TEST_ROOT}/test_mix_opto_gain_fast.cpp" APPEND PROPERTY COMPILE_OPTIONS -ffast-math)
    endif()
    add_test(NAME adi_mix_opto_gain_tests COMMAND adi_mix_opto_gain_tests)
endfunction()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    adi_add_opto_gain_tests()
else()
    set(ADI_OPTO_GAIN_TEST_ROOT "${ADI_OPTO_GAIN_TEST_ROOT}" PARENT_SCOPE)
endif()
