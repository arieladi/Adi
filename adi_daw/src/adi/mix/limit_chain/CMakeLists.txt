if(TARGET adi_mix_limit_chain)
    return()
endif()
if(NOT TARGET adi_mix_clip)
    add_subdirectory(../clip "${CMAKE_CURRENT_BINARY_DIR}/clip")
endif()
if(NOT TARGET adi_mix_limiter)
    add_subdirectory(../limiter "${CMAKE_CURRENT_BINARY_DIR}/limiter")
endif()
add_library(adi_mix_limit_chain STATIC Core.cpp Backend.cpp)
set_target_properties(adi_mix_limit_chain PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adi_mix_limit_chain PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
target_compile_features(adi_mix_limit_chain PUBLIC cxx_std_20)
target_link_libraries(adi_mix_limit_chain PRIVATE adi_mix_clip adi_mix_limiter)
if(TARGET adi_warnings)
    target_link_libraries(adi_mix_limit_chain PRIVATE adi_warnings)
endif()
if(MSVC)
    target_compile_options(adi_mix_limit_chain PRIVATE /fp:precise /utf-8)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        target_compile_options(adi_mix_limit_chain PRIVATE /clang:-ffp-model=precise /clang:-ffp-contract=off)
    endif()
else()
    target_compile_options(adi_mix_limit_chain PRIVATE -ffp-contract=off)
endif()
